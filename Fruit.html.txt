    <!-- Script JavaScript untuk Logika Aplikasi & Penyimpanan Lokal -->
    <script>
        // Data Awal (Contoh)
        let orders = JSON.parse(localStorage.getItem('fruit_orders')) || [
            { nama: "Mangga Harum Manis", jumlah: 29, modal: 6000, jual: 7000 },
            { nama: "Apel Merah Fuji", jumlah: 12, modal: 7000, jual: 8500 },
            { nama: "Pir Century", jumlah: 10, modal: 7000, jual: 8500 },
            { nama: "Buah Naga", jumlah: 9, modal: 7000, jual: 8500 },
            { nama: "Pisang Cavendish", jumlah: 5, modal: 6000, jual: 7500 }
        ];

        function formatRupiah(angka) {
            return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', minimumFractionDigits: 0 }).format(angka);
        }

        function renderDashboard() {
            const tableBody = document.getElementById('tableBody');
            tableBody.innerHTML = '';

            let totalPesanan = 0;
            let totalModalAll = 0;
            let totalOmsetAll = 0;
            let totalLabaAll = 0;

            orders.forEach((item, index) => {
                let totalModal = item.jumlah * item.modal;
                let totalOmset = item.jumlah * item.jual;
                let laba = totalOmset - totalModal;

                totalPesanan += Number(item.jumlah);
                totalModalAll += totalModal;
                totalOmsetAll += totalOmset;
                totalLabaAll += laba;

                let row = `
                    <tr class="hover:bg-gray-50">
                        <td class="p-3 border text-center">${index + 1}</td>
                        <td class="p-3 border font-medium text-gray-800">${item.nama}</td>
                        <td class="p-3 border text-center">${item.jumlah}</td>
                        <td class="p-3 border">${formatRupiah(item.modal)}</td>
                        <td class="p-3 border">${formatRupiah(item.jual)}</td>
                        <td class="p-3 border">${formatRupiah(totalModal)}</td>
                        <td class="p-3 border">${formatRupiah(totalOmset)}</td>
                        <td class="p-3 border font-semibold text-green-600">${formatRupiah(laba)}</td>
                        <td class="p-3 border text-center">
                            <button onclick="hapusData(${index})" class="bg-red-500 text-white px-2 py-1 rounded text-xs hover:bg-red-600">Hapus</button>
                        </td>
                    </tr>
                `;
                tableBody.innerHTML += row;
            });

            // Update KPI Cards
            document.getElementById('cardTotalPesanan').innerText = totalPesanan + " Pack";
            document.getElementById('cardTotalModal').innerText = formatRupiah(totalModalAll);
            document.getElementById('cardTotalOmset').innerText = formatRupiah(totalOmsetAll);
            document.getElementById('cardTotalLaba').innerText = formatRupiah(totalLabaAll);

            // Simpan ke localStorage
            localStorage.setItem('fruit_orders', JSON.stringify(orders));
        }

        document.getElementById('orderForm').addEventListener('submit', function(e) {
            e.preventDefault();
            const nama = document.getElementById('namaBuah').value;
            const jumlah = document.getElementById('jumlah').value;
            const modal = document.getElementById('hargaSupplier').value;
            const jual = document.getElementById('hargaJual').value;

            orders.push({ nama, jumlah, modal, jual });
            renderDashboard();

            this.reset();
        });

        function hapusData(index) {
            if(confirm('Yakin ingin menghapus data ini?')) {
                orders.splice(index, 1);
                renderDashboard();
            }
        }

        // Jalankan saat pertama buka
        renderDashboard();
    </script>
